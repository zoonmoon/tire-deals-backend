console.log("ProductGridAndFilters loader script has loaded!"); // ✅ logs when script is parsed


import React from "react";
import { createRoot } from "react-dom/client";

import ProductGridAndFilters from "../_frontend-components/product-grid-and-filters";

let root = null;

function loadProductGridAndFilters(collectionID, searchQuery, wheelAndTireSpecsArr) {
  const container = document.getElementById("product-grid-and-filters-wrapper");
  if (!container) {
    console.error("Missing #product-grid-and-filters-wrapper element in DOM");
    return;
  }

  if (!root) {
    root = createRoot(container);
  }

  root.render(
    <ProductGridAndFilters
      key={Date.now()}
      endpoint={'https://midwest-takeoffs-search-and-filter.vercel.app/api/storefront/fetch'}
      collectionID={collectionID}
      searchQuery={searchQuery}
      wheelAndTireSpecsArr={wheelAndTireSpecsArr}
    />
  );
}

window.loadProductGridAndFilters = loadProductGridAndFilters;
